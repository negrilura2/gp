```python
def verify(self, wav_path: str, threshold: float = None) -> Dict[str, Any]:
    thr = threshold if threshold is not None else DEFAULT_THRESHOLD
    emb = extract_embedding(
        self.model,
        self.device,
        wav_path,
        feature_type=self.feature_type,
        n_mels=self.n_mels
    )
    best_spk, best_score = None, -1.0
    if self.vector_store and self.vector_store.count() > 0:
        matches = self.vector_store.search(emb, top_k=1)
        if matches:
            best_spk, best_score = matches[0]
    result = "ACCEPT" if best_score >= thr else "REJECT"
    return {
        "status": "success",
        "result": result,
        "best_score": float(best_score),
        "best_speaker": best_spk,
        "predicted_user": best_spk,
        "threshold": thr
    }
```

## 2. ChromaDB 声纹向量检索逻辑

来源：`03_code_sources/ai_backend/vector_store.py`

```python
def search(self, query_embedding: np.ndarray, top_k: int = 1) -> List[Tuple[str, float]]:
    query_embedding = query_embedding / (np.linalg.norm(query_embedding) + 1e-9)
    results = self._collection.query(
        query_embeddings=[query_embedding.tolist()],
        n_results=top_k,
        include=["metadatas", "distances"]
    )
    ids = results["ids"][0]
    distances = results["distances"][0]
    matches = []
    for uid, dist in zip(ids, distances):
        sim = 1.0 - dist
        matches.append((uid, float(sim)))
    return matches
```

```python
class HttpVoiceService:
    def verify(self, wav_path, threshold=None):
        files = []
        f = open(wav_path, "rb")
        files.append(("file", (os.path.basename(wav_path), f, "audio/wav")))
```

```javascript
const wsUrl = "ws://localhost:9000/ws/audio";
ws = new WebSocket(wsUrl);

ws.onmessage = (event) => {
  const data = JSON.parse(event.data);
  if (data.type === "result_start") {
    agentResult.value = {
      identity: data.identity,
      text: data.text,
      agent: { response: "" }
    };
  } else if (data.type === "agent_chunk") {
    agentResult.value.agent.response += data.text;
  } else if (data.type === "tts_chunk") {
    handleAudioChunk(data.audio);
  }
};
```
